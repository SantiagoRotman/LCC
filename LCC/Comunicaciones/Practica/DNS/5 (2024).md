## dns.rubikia.com
etc/bind/named.conf
~~~ 
view "trabajo" {
	match-clients {192.168.7.0/24};
	zone "rubikia.com" {
		type master;
		allow-transfer {slaves;};
		file "etc/bind/db.rubikia";
	};
	zone "7.168.192.in-addr.arpa" {
		type master;
		file "etc/bind/inv.rubikia";
	};
	zone "spiralti.com" {
		type slave;
		masters {31.13.77.51};
		file "etc/bind/db.spiralti";
	};
};

view "invitados" {
	match-clients {192.168.8.0/24};
	zone "rubikia.com" {
		type master;
		file "etc/bind/db.rubikia-invitados";
	};
};

view "external" {
	match-clients {any;};
	zone "rubikia.com" {
		type master;
		file "etc/bind/db.rubikia";
	};
	zone "7.168.192.in-addr.arpa" {
		type master;
		file "etc/bind/inv.rubikia";
	};
	zone "spiralti.com" {
		type slave;
		masters {31.13.77.51};
		file "etc/bind/db.spiralti";
	};
};
~~~

## dns2.ddi.rubikia.com
etc/bind/named.conf
~~~
view "trabajo" {
	match-clients {192.168.7.0/24; 127.0.0.0/8};
	zone "ddi.rubika.com" {
		type master;
		file "etc/bind/db.ddi"
	};
	zone "rubika.com" {
		type slave;
		masters {192.168.7.180;};
		file "etc/bind/db.rubikia"
	};
};
~~~


## dns.rubikia.com
etc/bind/db.rubikia en dns o dns2

las lines que tienen un * no deberian ir en db.rubikia-internal
~~~
$ORIGIN rubikia.com.
$TTL 2d

@               IN       SOA               dns hostmaster (12h 15m 3w 3h)
                IN       NS                dns.rubikia.com.
                IN       NS                dns2.ddi.rubikia.com.
*               IN       NS                nsbackup.rubikia.com.
                IN       MX      10        mails.rubikia.com.
                
dns             IN       A                 192.168.7.18
mails           IN       CNAME             dns
www             IN       A                 192.168.7.181
*nsbackup       IN       A                 193.168.7.18

ddi             IN       NS                dns2.ddi
dns2.ddi        IN       A                 192.168.7.180
~~~

etc/bind/inv.rubikia
~~~

$ORIGIN 7.168.192.in-addr.arpa.
$TTL 2d

@               IN       SOA               dns hostmaster (12h 15m 3w 3h)
				IN       NS                dns
				IN       MX        10      mails

18              IN       PTR               dns
180             IN       PTR               dns2.ddi
181             IN       PTR               www
~~~
